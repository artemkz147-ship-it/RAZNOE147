  buildPhysics() {
    const groundBody = this.world.createRigidBody(RAPIER.RigidBodyDesc.fixed().setTranslation(0, -0.3, 0));
    this.world.createCollider(RAPIER.ColliderDesc.cuboid(210, 0.25, 210).setFriction(1.0), groundBody);
    for (let i = 0; i < this.trackSamples.length; i += 5) {
      const p = this.trackSamples[i], t = this.trackTangents[i];
      const n = new THREE.Vector3(-t.z, 0, t.x); const yaw = Math.atan2(-t.x, -t.z);
      const rot = { x: 0, y: Math.sin(yaw / 2), z: 0, w: Math.cos(yaw / 2) };
      for (const side of [-1, 1]) {
        const bp = p.clone().addScaledVector(n, side * (this.roadWidth * 0.5 + 0.75));
        const rb = this.world.createRigidBody(RAPIER.RigidBodyDesc.fixed().setTranslation(bp.x, 0.62, bp.z).setRotation(rot));
        this.world.createCollider(RAPIER.ColliderDesc.cuboid(2.9, 0.7, 0.2).setFriction(0.45).setRestitution(0.08), rb);
      }
    }
  }

  spawnDestructibles() {
    const barrierMat = new THREE.MeshStandardMaterial({ color: 0xff7138, emissive: 0x5a1907, emissiveIntensity: .55, metalness: .15, roughness: .62 });
    const crateMat = new THREE.MeshStandardMaterial({ color: 0xc28b45, roughness: .82, metalness: .04 });
    const bollardMat = new THREE.MeshStandardMaterial({ color: 0x44eaff, emissive: 0x087691, emissiveIntensity: .9, metalness: .4, roughness: .36 });

    for (let i = 0; i < 48; i++) {
      const kind: DestructibleKind = i % 3 === 0 ? 'barrier' : i % 3 === 1 ? 'crate' : 'bollard';
      const t = mod(.018 + i * .0199, 1); const p = this.track.getPointAt(t); const tan = this.track.getTangentAt(t).normalize(); const n = new THREE.Vector3(-tan.z, 0, tan.x);
      const side = i % 2 ? 1 : -1; const offset = this.roadWidth * .5 + (kind === 'barrier' ? 2.1 : 3.5 + (i % 4));
      const pos = p.clone().addScaledVector(n, side * offset);
      const yaw = Math.atan2(-tan.x, -tan.z) + (kind === 'crate' ? i * .37 : 0);
      let mesh: THREE.Mesh; let half = new THREE.Vector3();
      if (kind === 'barrier') { mesh = new THREE.Mesh(new THREE.BoxGeometry(2.3, .78, .55), barrierMat.clone()); half.set(1.15, .39, .275); pos.y = .43; }
      else if (kind === 'crate') { mesh = new THREE.Mesh(new THREE.BoxGeometry(1.05, 1.05, 1.05), crateMat.clone()); half.set(.525, .525, .525); pos.y = .56; }
      else { mesh = new THREE.Mesh(new THREE.CylinderGeometry(.16, .22, 2.5, 10), bollardMat.clone()); half.set(.22, 1.25, .22); pos.y = 1.27; }
      mesh.castShadow = true; mesh.receiveShadow = true; mesh.position.copy(pos); mesh.rotation.y = yaw; this.scene.add(mesh);
      const rot = new THREE.Quaternion().setFromAxisAngle(Y_AXIS, yaw);
      const body = this.world.createRigidBody(RAPIER.RigidBodyDesc.dynamic().setTranslation(pos.x, pos.y, pos.z).setRotation({ x: rot.x, y: rot.y, z: rot.z, w: rot.w }).setLinearDamping(.3).setAngularDamping(.35).setCanSleep(true));
      this.world.createCollider(RAPIER.ColliderDesc.cuboid(half.x, half.y, half.z).setDensity(kind === 'bollard' ? 14 : 45).setFriction(.72).setRestitution(.24), body);
      this.destructibles.push({ mesh, body, initialPosition: pos.clone(), initialRotation: rot.clone(), kind });
    }
  }
